# 数据模型（Postgres 16）

原则：**事实表只追加不改**（事件溯源式留证），状态表可以 UPDATE 但必须有 `updated_at` 与版本列。所有时间 `timestamptz`，存 UTC。

```sql
-- ============ 租户与账号 ============
CREATE TABLE users (
  id             bigserial PRIMARY KEY,
  email          citext UNIQUE,
  phone          text UNIQUE,
  password_hash  text,                          -- argon2id；扫码用户可空
  role           text NOT NULL DEFAULT 'user',  -- user | ops | admin
  status         text NOT NULL DEFAULT 'active',-- active | frozen | deleted
  balance_cents  bigint NOT NULL DEFAULT 0,
  created_at     timestamptz NOT NULL DEFAULT now(),
  updated_at     timestamptz NOT NULL DEFAULT now()
);

-- 雨课堂账号（一个 user 可绑多个）
CREATE TABLE accounts (
  id                bigserial PRIMARY KEY,
  user_id           bigint NOT NULL REFERENCES users(id),
  domain            text NOT NULL,              -- www/pro/changjiang/huanghe.yuketang.cn
  ykt_user_id       text NOT NULL,              -- 雨课堂 UserID
  nickname          text,
  school            text,
  cookie_cipher     bytea NOT NULL,             -- AES-256-GCM(cookie)
  dek_wrapped       bytea NOT NULL,             -- AES-256-GCM(DEK)
  key_version       int  NOT NULL DEFAULT 1,
  session_expires_at timestamptz,               -- 登录时 +14d
  status            text NOT NULL DEFAULT 'active', -- active|expired|banned|paused
  proxy_id          bigint,                     -- 出口代理（粘性）
  created_at        timestamptz NOT NULL DEFAULT now(),
  updated_at        timestamptz NOT NULL DEFAULT now(),
  UNIQUE (domain, ykt_user_id)
);
CREATE INDEX ON accounts (user_id, status);
CREATE INDEX ON accounts (session_expires_at) WHERE status = 'active';

-- ============ 订单与计费 ============
CREATE TABLE orders (
  id             bigserial PRIMARY KEY,
  user_id        bigint NOT NULL REFERENCES users(id),
  account_id     bigint REFERENCES accounts(id),
  plan           text NOT NULL,                 -- trial|week|month|term
  amount_cents   bigint NOT NULL,
  status         text NOT NULL DEFAULT 'pending',-- pending|paid|refunded|closed
  provider       text,                          -- wechat|alipay|manual
  provider_txn   text,
  paid_at        timestamptz,
  expire_at      timestamptz NOT NULL,          -- 服务到期
  created_at     timestamptz NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX ON orders (provider, provider_txn) WHERE provider_txn IS NOT NULL;

-- 计次流水（成功签到才计次，失败不计）
CREATE TABLE billing_events (
  id          bigserial PRIMARY KEY,
  account_id  bigint NOT NULL REFERENCES accounts(id),
  order_id    bigint REFERENCES orders(id),
  kind        text NOT NULL,                    -- checkin|quiz
  lesson_id   bigint,
  charged     boolean NOT NULL DEFAULT true,
  created_at  timestamptz NOT NULL DEFAULT now()
);

-- ============ 课表与课堂 ============
CREATE TABLE course_schedules (
  id             bigserial PRIMARY KEY,
  account_id     bigint NOT NULL REFERENCES accounts(id),
  classroom_id   bigint NOT NULL,
  classroom_name text NOT NULL,
  weekday        smallint,                      -- 1-7；NULL 表示一次性
  start_time     time,
  end_time       time,
  start_date     date,
  end_date       date,
  rrule          text,                          -- 或 RFC5545，二选一
  enabled        boolean NOT NULL DEFAULT true,
  config         jsonb NOT NULL DEFAULT '{}',   -- 覆盖全局策略：是否答题/位置/暗号
  created_at     timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE lessons (
  id             bigserial PRIMARY KEY,
  account_id     bigint NOT NULL REFERENCES accounts(id),
  domain         text NOT NULL,
  lesson_id      bigint NOT NULL,
  classroom_id   bigint,
  title          text,
  teacher        text,
  started_at     timestamptz,
  ended_at       timestamptz,
  status         text NOT NULL DEFAULT 'pending', -- pending|warming|live|ended|failed
  lesson_token   text,                            -- wss auth，短命，随 lesson 存
  identity_id    text,
  trace_key      text,                            -- MinIO 对象前缀
  created_at     timestamptz NOT NULL DEFAULT now(),
  UNIQUE (account_id, lesson_id)
);

-- ============ 任务 ============
CREATE TABLE tasks (
  id            bigserial PRIMARY KEY,
  account_id    bigint NOT NULL REFERENCES accounts(id),
  lesson_row_id bigint REFERENCES lessons(id),
  kind          text NOT NULL,                  -- warmup|checkin|quiz|keepalive
  idempotency   text NOT NULL,                  -- acct:lesson:kind[:problemId]
  payload       jsonb NOT NULL DEFAULT '{}',
  deadline_at   timestamptz,
  status        text NOT NULL DEFAULT 'queued', -- queued|running|done|failed|skipped
  attempts      smallint NOT NULL DEFAULT 0,
  last_error    text,
  created_at    timestamptz NOT NULL DEFAULT now(),
  updated_at    timestamptz NOT NULL DEFAULT now(),
  UNIQUE (idempotency)                          -- 幂等兜底：最多一次签到
);
CREATE INDEX ON tasks (status, deadline_at) WHERE status IN ('queued','running');

CREATE TABLE task_runs (
  id          bigserial PRIMARY KEY,
  task_id     bigint NOT NULL REFERENCES tasks(id),
  agent_id    text NOT NULL,
  step        text NOT NULL,                    -- enter_checkin|code_checkin|dyn_checkin|geo_checkin
  ok          boolean NOT NULL,
  http_status int,
  ykt_msg     text,                             -- OK / LESSON_END / ...
  duration_ms int,
  trace_key   text,
  created_at  timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON task_runs (task_id, created_at);

-- ============ 题库 ============
CREATE TABLE questions (
  id            bigserial PRIMARY KEY,
  domain        text NOT NULL,
  problem_key   text NOT NULL,                  -- problemId 或 题干 sha256
  type          text NOT NULL,                  -- single|multi|blank|vote
  body          text NOT NULL,
  options       jsonb,
  answer        jsonb,
  confidence    real NOT NULL DEFAULT 0,
  source        text NOT NULL,                  -- official|shared|llm|manual
  hits          int NOT NULL DEFAULT 0,
  created_at    timestamptz NOT NULL DEFAULT now(),
  UNIQUE (domain, problem_key)
);
CREATE INDEX ON questions USING gin (to_tsvector('simple', body));

-- ============ 协议 profile ============
CREATE TABLE domain_profiles (
  domain            text PRIMARY KEY,
  protocol_version  text NOT NULL,
  checkin_source    int NOT NULL,               -- 5/14/21，实测确定
  headers           jsonb NOT NULL,
  field_map         jsonb NOT NULL,
  supports_geo      boolean NOT NULL DEFAULT false,
  supports_dynamic  boolean NOT NULL DEFAULT false,
  enabled           boolean NOT NULL DEFAULT true,
  note              text,
  updated_at        timestamptz NOT NULL DEFAULT now()
);

-- ============ 出口代理 ============
CREATE TABLE proxies (
  id         bigserial PRIMARY KEY,
  url        text NOT NULL,                     -- socks5://user:pass@host:port
  region     text,
  status     text NOT NULL DEFAULT 'ok',        -- ok|cooling|dead
  fail_count int NOT NULL DEFAULT 0,
  last_ok_at timestamptz,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- ============ 审计 ============
CREATE TABLE audit_logs (
  id         bigserial PRIMARY KEY,
  actor_id   bigint,
  actor_role text,
  action     text NOT NULL,                     -- account.unbind|task.retry|config.update
  target     text,
  detail     jsonb NOT NULL DEFAULT '{}',
  ip         inet,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON audit_logs (created_at DESC);
```

## 迁移规范
- 全部迁移放 `deploy/migrations/`，命名 `0001_init.up.sql` / `0001_init.down.sql`。
- **[M]** 每个迁移必须可回滚；不可回滚的（如删列）必须拆两步发版（先停用，再删）。
- **[M]** 大表加索引用 `CREATE INDEX CONCURRENTLY`，且单独一个迁移文件。
- 保留策略：`task_runs` 90 天分区归档，`audit_logs` 永久，MinIO 留证 7 天。
